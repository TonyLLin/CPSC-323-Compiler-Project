/* program 3: compute area and perimeter of a rectangle
   uses two helper functions and a menu-style loop */
@
function area (w integer, h integer)
{
    return w * h;
}

function perimeter (w integer, h integer)
{
    return 2 * w + 2 * h;
}
@
integer width, height, choice;
boolean running;
@
running = true;
read (width);
read (height);

while (running == true)
{
    read (choice);
    if (choice == 1)
        write (area (width, height));
    fi
    if (choice == 2)
        write (perimeter (width, height));
    fi
    if (choice == 3)
        running = false;
    fi
}

write (running);
@
